# CIトリガー
# 1 必ずリポジトリ直下へ移動
cd "C:\projects\Process ReDesigner"

# 2 いまどこにいるか確認（念のため）
git rev-parse --show-toplevel

# 3 変更確認
git status --short

# 4 必要ファイルだけを明示的にステージ（git add . は使わない）
#    ※ rdf -> prd のリネームを含むため、削除/追加をまとめてステージ
git add deploy.txt
git add -A .github/workflows
git add -A rdf prd

# 5 ステージ内容を最終確認（ここが超重要）
git diff --cached --name-only

# 6 コミット
git commit -m "chore(prd): unify frontend deploy flow"

# 7 プッシュ
git push origin HEAD
